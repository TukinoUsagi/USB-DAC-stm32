/**
  ******************************************************************************
  * @file    usbd_audio_if.c
  * @brief   USB Audio再生エンジン本体
  *
  * ----------------------------------------------------------------------------
  * 動作の流れ
  * ----------------------------------------------------------------------------
  *   PC (USB Audio Class 1.0, 16bit stereo, USBD_AUDIO_FREQ Hz)
  *       ↓ (毎USBフレーム=1msごとにISOアウト転送)
  *   usbd_audio.c (ST公式ミドルウェア)
  *       ↓ haudio->buffer[] というリングバッファに書き込み
  *       ↓
  *   USB側 : haudio->wr_ptr
  *       ↓
  *   TIM4 割り込み (USBD_AUDIO_FREQ Hzごとに発火)
  *       ↓
  *   TIM4側 : haudio->rd_ptr
  *       ↓
  *   PCM 1サンプル(L16+R16=4byte)取り出し
  *       ↓
  *   0..65535 にオフセットしてPWM CCR範囲にスケーリング
  *       ↓
  *   TIM1 CCR1(LEFT) / CCR2(RIGHT) 書き込み
  *       ↓
  *   TIM1 120kHz相補PWM出力 (PA8/PB13, PA9/PB14)
  *
  * ----------------------------------------------------------------------------
  * リングバッファ管理
  * ----------------------------------------------------------------------------
  *   USB受信側 : haudio->wr_ptr を進める
  *   TIM4側    : haudio->rd_ptr を進める
  *
  *   rd_ptr と wr_ptr は独立して動作する。
  *   TIM4側では独自の再生ポインタを持たず、
  *   ST USB Audioミドルウェア本来の haudio->rd_ptr を使用する。
  *
  ******************************************************************************

#include "main.h"


/* ----------------------------------------------------------------------------
 * PWM CCR 可動範囲
 *
 * TIM1 Period=599
 * DeadTime=7
 * ----------------------------------------------------------------------------*/
#define PWM_CCR_MIN   30U
#define PWM_CCR_MAX   570U

/* UART送信は割込み内で行わず、8サンプル平均をmain()へ渡す。 */
#define PCM_DEBUG_AVERAGE_SAMPLES  8U
#define PCM_DEBUG_QUEUE_SIZE       64U

extern TIM_HandleTypeDef htim1;
TIM_HandleTypeDef htim4;

/*
 * USB Audio再生中フラグ
 */
volatile uint8_t g_usb_audio_playing = 0U;

/*
 * デバッグ用PCMキュー
 */
static volatile int16_t g_debug_left[PCM_DEBUG_QUEUE_SIZE];
static volatile int16_t g_debug_right[PCM_DEBUG_QUEUE_SIZE];
static volatile uint8_t g_debug_write = 0U;
static volatile uint8_t g_debug_read = 0U;
static int32_t g_debug_left_sum = 0;
static int32_t g_debug_right_sum = 0;
static uint8_t g_debug_sample_count = 0U;

/*
 * USBD Audio interface callback

static int8_t AUDIO_IF_Init(uint32_t AudioFreq,
                            uint32_t Volume,

static int8_t AUDIO_IF_VolumeCtl(uint8_t vol);

/**
  * @brief  無音(PWM中央値)をCCRに書き込む
  */
static void AUDIO_OutputSilence(void)
{
  uint32_t mid = (PWM_CCR_MIN + PWM_CCR_MAX) / 2U;

  __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_1, mid);
  __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_2, mid);
}


/**
  * @brief  AUDIO Interface Init
  */
static int8_t AUDIO_IF_Init(uint32_t AudioFreq,
                            uint32_t Volume,
                            uint32_t options)
{
  UNUSED(AudioFreq);
  UNUSED(Volume);
  UNUSED(options);

  AUDIO_OutputSilence();

  return 0;
}


/**
  * @brief  AUDIO Interface DeInit
  */
static int8_t AUDIO_IF_DeInit(uint32_t options)
{
  UNUSED(options);

  g_usb_audio_playing = 0U;

  HAL_TIM_Base_Stop_IT(&htim4);

  AUDIO_OutputSilence();

  return 0;
}


/**
  * @brief  AUDIO_CMD_START / PLAY / STOP を処理する
  *
  *   START:
  *       USB側でリングバッファへの初期データ蓄積が完了。
 *       TIM4を起動する。
  *
  *       pbufは保存しない。
  *       TIM4側はUSBD_AUDIO_HandleTypeDefの
  *       haudio->buffer[] と haudio->rd_ptr を直接使用する。
  *
  *   PLAY:
  *       再同期チェックポイント。
  *       rd_ptrはここでは変更しない。
  *
  *   STOP:
  *       TIM4を停止して無音を出力する。
  */
static int8_t AUDIO_IF_AudioCmd(uint8_t *pbuf,
                                uint32_t size,
                                uint8_t cmd)
{
  UNUSED(pbuf);
  UNUSED(size);

  switch (cmd)
  {
    case AUDIO_CMD_START:

      /*
       * rd_ptrはST USB Audioミドルウェア側の
       * haudio->rd_ptrをTIM4が直接使用する。
       *
       * ここでは独自の再生ポインタを持たない。
       */

      g_debug_write = 0U;
      g_debug_read = 0U;
      g_debug_left_sum = 0;
      g_debug_right_sum = 0;
      g_debug_sample_count = 0U;

      g_usb_audio_playing = 1U;

      HAL_TIM_Base_Start_IT(&htim4);

      break;


    case AUDIO_CMD_PLAY:

      /*
       * rd_ptrはTIM4側だけが進める。
       *
       * USBD_AUDIO_Sync()から呼ばれても
       * rd_ptrを変更しない。
       */

      break;


    case AUDIO_CMD_STOP:

      g_usb_audio_playing = 0U;

      HAL_TIM_Base_Stop_IT(&htim4);

      AUDIO_OutputSilence();

      break;


    default:

      break;
  }

  return 0;
}

  * @brief  Volume control

/**
  * @brief  デバッグキューからPCM平均値を取り出す
  */
uint8_t AUDIO_DebugPop(int16_t *left, int16_t *right)
{
  uint8_t read = g_debug_read;

  if (read == g_debug_write)
  {
    return 0U;
  }

  *left = g_debug_left[read];
  *right = g_debug_right[read];

  g_debug_read =
      (uint8_t)((read + 1U) % PCM_DEBUG_QUEUE_SIZE);

  return 1U;
}


/* ============================================================================
 * TIM4 : オーディオサンプルレート割り込みタイマー
 *
 *   TIM4 = USBD_AUDIO_FREQ Hz
 *
 *   例:
 *     USBD_AUDIO_FREQ = 16000Hz
 *     TIM4 clock      = 72MHz
 *     ARR             = 4499
 *
 *   72MHz / (4499 + 1) = 16000Hz
 * ==========================================================================*/

void MX_TIM4_Init(void)
{
  TIM_ClockConfigTypeDef sClockSourceConfig = {0};
  TIM_MasterConfigTypeDef sMasterConfig = {0};

  uint32_t arr =
      (72000000U / USBD_AUDIO_FREQ) - 1U;

  htim4.Instance               = TIM4;
  htim4.Init.Prescaler         = 0;
  htim4.Init.CounterMode       = TIM_COUNTERMODE_UP;
  htim4.Init.Period            = arr;
  htim4.Init.ClockDivision     = TIM_CLOCKDIVISION_DIV1;
  htim4.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_ENABLE;

  if (HAL_TIM_Base_Init(&htim4) != HAL_OK)
  {
    Error_Handler();
  }

  sClockSourceConfig.ClockSource = TIM_CLOCKSOURCE_INTERNAL;

  if (HAL_TIM_ConfigClockSource(&htim4,
                                &sClockSourceConfig) != HAL_OK)
  {
    Error_Handler();
  }

  sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
  sMasterConfig.MasterSlaveMode     = TIM_MASTERSLAVEMODE_DISABLE;

  if (HAL_TIMEx_MasterConfigSynchronization(
          &htim4,
          &sMasterConfig) != HAL_OK)
  {
    Error_Handler();
  }

  HAL_NVIC_SetPriority(TIM4_IRQn, 2, 0);
  HAL_NVIC_EnableIRQ(TIM4_IRQn);

  /*
   * HAL_TIM_Base_Start_IT()は
   * AUDIO_CMD_START受信時に実行する。
   */
}


extern USBD_HandleTypeDef hUsbDeviceFS;


/**
  * @brief  TIM周期割り込みコールバック
  *
  * 重要:
  *
  *   TIM4側では独自のg_play_ptrを使用しない。
  *
  *   ST USB Audio middlewareが持っている
  *
  *       haudio->rd_ptr
  *       haudio->buffer[]
  *
  *   を直接使用する。
  *
  *   1回の割り込みで1 PCM stereo sample = 4byteを消費する。
  */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  USBD_AUDIO_HandleTypeDef *haudio;

  if (htim->Instance != TIM4)
  {
    return;
  }

  if (g_usb_audio_playing == 0U)
  {
    return;
  }

  /*
   * ST USB Audio middlewareのハンドルを取得。
   */
  if (hUsbDeviceFS.pClassDataCmsit[hUsbDeviceFS.classId] == NULL)
  {
    return;
  }

  haudio =
      (USBD_AUDIO_HandleTypeDef *)
      hUsbDeviceFS.pClassDataCmsit[hUsbDeviceFS.classId];

  /*
   * rd_ptrはTIM4側だけが進める。
   */
  uint16_t rd_ptr = haudio->rd_ptr;

  /*
   * リングバッファ範囲チェック。
   *
   * 1 stereo sample = 4byteなので、
   * 最後の4byteを読み出せる位置だけを使用する。
   */
  if (rd_ptr > (AUDIO_TOTAL_BUF_SIZE - 4U))
  {
    rd_ptr = 0U;
    haudio->rd_ptr = 0U;
  }

  /*
   * PCM16 stereo
   *
   * buffer[rd_ptr + 0] = Left Low
   * buffer[rd_ptr + 1] = Left High
   * buffer[rd_ptr + 2] = Right Low
   * buffer[rd_ptr + 3] = Right High
   */

  int16_t left =
      (int16_t)((uint16_t)haudio->buffer[rd_ptr] |
                ((uint16_t)haudio->buffer[rd_ptr + 1U] << 8));

  int16_t right =
      (int16_t)((uint16_t)haudio->buffer[rd_ptr + 2U] |
                ((uint16_t)haudio->buffer[rd_ptr + 3U] << 8));

  /*
   * 1 stereo sample = 4byte消費
   */
  rd_ptr += 4U;

  if (rd_ptr >= AUDIO_TOTAL_BUF_SIZE)
  {
    rd_ptr = 0U;
  }

  /*
   * 本来のrd_ptrへ書き戻す。
   *
   * これ以外の場所ではTIM4の再生位置を変更しない。
   */
  haudio->rd_ptr = rd_ptr;


  /*
   * signed 16bit
   *
   *   -32768 ～ +32767
   *
   * ↓
   *
   * unsigned
   *
   *   0 ～ 65535
   *
   * ↓
   *
   * PWM CCR
   *
   *   30 ～ 570
   */

  uint32_t uleft =
      (uint32_t)((int32_t)left + 32768);

  uint32_t uright =
      (uint32_t)((int32_t)right + 32768);

  uint32_t ccrL =
      PWM_CCR_MIN +
      (uleft * (PWM_CCR_MAX - PWM_CCR_MIN)) / 65535U;

  uint32_t ccrR =
      PWM_CCR_MIN +
      (uright * (PWM_CCR_MAX - PWM_CCR_MIN)) / 65535U;


  /*
   * TIM1 PWM duty更新
   */
  __HAL_TIM_SET_COMPARE(&htim1,
                        TIM_CHANNEL_1,
                        ccrL);

  __HAL_TIM_SET_COMPARE(&htim1,
                        TIM_CHANNEL_2,
                        ccrR);


  /*
   * PCMデバッグ用
   *
   * 8サンプル平均してmain()側へ渡す。
   */
  g_debug_left_sum += left;
  g_debug_right_sum += right;
  g_debug_sample_count++;

  if (g_debug_sample_count == PCM_DEBUG_AVERAGE_SAMPLES)
  {
    uint8_t next =
        (uint8_t)((g_debug_write + 1U) %
                  PCM_DEBUG_QUEUE_SIZE);

    if (next != g_debug_read)
    {
      g_debug_left[g_debug_write] =
          (int16_t)(g_debug_left_sum /
                    (int32_t)PCM_DEBUG_AVERAGE_SAMPLES);

      g_debug_right[g_debug_write] =
          (int16_t)(g_debug_right_sum /
                    (int32_t)PCM_DEBUG_AVERAGE_SAMPLES);

      g_debug_write = next;
    }

    g_debug_left_sum = 0;
    g_debug_right_sum = 0;
    g_debug_sample_count = 0U;
  }
}

# 2. `Middlewares/ST/STM32_USB_Device_Library/Class/AUDIO/Src/usbd_audio.c`

こちらは、**`USBD_AUDIO_Sync()` が `rd_ptr` を動かさない**ようにします。

現在のこの部分、

```c
if (haudio->rd_enable == 1U)
{
  haudio->rd_ptr += (uint16_t)BufferSize;

  if (haudio->rd_ptr == AUDIO_TOTAL_BUF_SIZE)
  {
    /* roll back */
    haudio->rd_ptr = 0U;
  }
}
```

は削除します。

さらに、その後の `BufferSize` の増減処理も、今回の構成では `rd_ptr` を動かすためには不要です。

したがって、**`USBD_AUDIO_Sync()` 全体を置き換える**のが一番安全です。

## 修正版 `USBD_AUDIO_Sync()`

```c
/**
  * @brief  USBD_AUDIO_Sync
  *         USB Audioバッファの同期チェックポイント
  *
  *  [変更]
  *  rd_ptrはTIM4側だけが管理する。
  *
  *  USB受信側:
  *      haudio->wr_ptr
  *
  *  TIM4再生側:
  *      haudio->rd_ptr
  *
  *  したがって、この関数ではrd_ptrを変更しない。
  *
  *  AUDIO_OFFSET_FULLの場合だけ、従来どおりAudioCmd()
  *  を呼び出して同期イベントを通知する。
  *
  * @param  pdev: device instance
  * @param  offset: audio offset
  * @retval None
  */
void USBD_AUDIO_Sync(USBD_HandleTypeDef *pdev,
                     AUDIO_OffsetTypeDef offset)
{
  USBD_AUDIO_HandleTypeDef *haudio;

  if (pdev->pClassDataCmsit[pdev->classId] == NULL)
  {
    return;
  }

  haudio =
      (USBD_AUDIO_HandleTypeDef *)
      pdev->pClassDataCmsit[pdev->classId];

  /*
   * offsetだけ更新する。
   *
   * rd_ptrは絶対に変更しない。
   */
  haudio->offset = offset;

  /*
   * AUDIO_OFFSET_FULLで同期イベントを通知。
   *
   * AudioCmd(AUDIO_CMD_PLAY)側もrd_ptrを変更しない。
   */
  if (haudio->offset == AUDIO_OFFSET_FULL)
  {
    ((USBD_AUDIO_ItfTypeDef *)
     pdev->pUserData[pdev->classId])->AudioCmd(
        &haudio->buffer[0],
        0U,
        AUDIO_CMD_PLAY);

    haudio->offset = AUDIO_OFFSET_NONE;
  }
}
```

# 5. 今回、特に消えるもの

`usbd_audio_if.c` から次の2つが消えます。

```c
static volatile uint8_t  *g_audio_ring = NULL;
static volatile uint32_t  g_play_ptr = 0U;
```

そして、

```c
g_audio_ring = pbuf;
g_play_ptr = 0U;
```

も消えます。

さらにTIM4の、

```c
g_audio_ring[g_play_ptr]
```

は完全になくなり、

```c
haudio->buffer[haudio->rd_ptr]
```

になります。

## ⚠️ ひとつだけ確認したい重要ポイント

この修正で**ポインタ管理の責任分担は完全に整理できます**。

ただし現在の `usbd_audio.c` では、`rd_enable` が

```c
if (haudio->rd_enable == 0U)
{
  if (haudio->wr_ptr == (AUDIO_TOTAL_BUF_SIZE / 2U))
  {
    haudio->rd_enable = 1U;
  }
}
```

として設定されています。

今回の設計では `rd_enable` は**TIM4の再生位置を動かす目的ではもう使いません**。

したがって次の段階では、

```text
rd_enable
offset
USBD_AUDIO_Sync()
```

を整理して、**「バッファに十分なPCMが入ったらTIM4再生開始」**という状態管理まで綺麗にするのが良いです。

まず今回は、**`rd_ptr` の二重管理をなくすところまで**に留めるのが安全です。
